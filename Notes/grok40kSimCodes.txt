def roll_dice(sides, num):
    return np.random.randint(1, sides + 1, num)

# Blessings patterns (simplified greedy: prioritize Lethal > Sust > Reroll Charge > Pile6 > FightDeath)
BLESSINGS = {
    'Warp Blades': {'pattern': [2,2,2], 'effect': 'lethal'},  # triple 2+
    'Martial Excellence': {'pattern': [4,4], 'effect': 'sust1'},  # double 4+
    'Unbridled Bloodlust': {'pattern': [2,2], 'effect': 'reroll_charge'},

def blessings_roll():
    roll = roll_dice(6, 8)
    roll.sort(reverse=True)
    active = {}
    used_dice = []
    # Greedy: Warp Blades first, then Martial, Bloodlust, etc.
    priorities = ['Warp Blades', 'Martial Excellence', 'Unbridled Bloodlust', 'Rage Fueled', 'Total Carnage']
    for b in priorities:
        pat = sorted(BLESSINGS[b]['pattern'], reverse=True)
        if len(roll) >= len(pat):
            match = True
            idx = 0
            for i in range(len(roll)):
                if idx < len(pat) and roll[i] >= pat[idx]:
                    idx += 1
                if idx == len(pat):
                    break
            if idx == len(pat):
                active[b] = BLESSINGS[b]['effect']
                # Remove used dice (simplified: remove matching)
                for p in pat:
                    for j in range(len(roll)):
                        if roll[j] >= p:
                            roll = np.delete(roll, j)
                            break
                break  # up to 2, but greedy one-by-one
        if len(active) == 2:
            break
    return active

def hit_rolls(a, bs, crit_on=6):
    rolls = roll_dice(6, a)
    crits = np.sum(rolls == crit_on)
    normals = np.sum(rolls >= bs)
    return normals + crits, crits

def wound_rolls(h, s, t, oath_bonus=0, reroll1=False):
    wound_req = 4 + (t > s) - (s > 2*t) + oath_bonus  # std 40k: 4 if =, 3 >, 2 double
    wound_req = max(2, min(6, wound_req))
    rolls = roll_dice(6, h)
    if reroll1:
        rolls[rolls == 1] = roll_dice(6, np.sum(rolls == 1))
    wounds = np.sum(rolls >= wound_req)
    return wounds

def save_rolls(w, ap, armour_save, invuln):
    save_req = max(2, armour_save + ap)
    p_save = 7 - save_req  # P success
    # Use invuln if better
    inv_req = invuln
    effective_req = min(save_req, inv_req)
    rolls = roll_dice(6, w)
    failed = np.sum(rolls < effective_req)
    return failed

def damage_rolls(failed, d_type):
    if d_type == 'fixed':
        return failed * 2  # D2 etc
    elif d_type == 'd4':
        return np.sum(roll_dice(4, failed))
    elif d_type == 'd6p2':
        return np.sum(roll_dice(6, failed) + 2)

def lion_pistol_damage(angron_w_left):
    if angron_w_left <= 0:
        return 0
    hits, crits = hit_rolls(LION_PISTOL_A, LION_PISTOL_BS)
    wounds = wound_rolls(hits, LION_PISTOL_S, ANGRON_T)
    failed = save_rolls(wounds, LION_PISTOL_AP, 2+? , ANGRON_INV)  # Angron armour 2+ Sv
    dmg = damage_rolls(failed, 'fixed')
    return min(dmg, angron_w_left)

def lion_strike_damage(angron_w_left, oath_bonus):
    hits, crits = hit_rolls(LION_STRIKE_A, LION_STRIKE_WS, crit_on=6)  # Lethal crit auto wound
    lethal_wounds = crits
    normal_wounds = wound_rolls(hits - crits, LION_STRIKE_S, ANGRON_T, oath_bonus)
    total_w = lethal_wounds + normal_wounds
    failed = save_rolls(total_w, LION_STRIKE_AP, 2, ANGRON_INV)
    dmg = damage_rolls(failed, 'd4')
    return min(dmg, angron_w_left)

def angron_strike_damage(lion_w_left, shield_active=True, lethal=False, sust=False):

def roll_d6(n=1):
    return np.random.randint(1, 7, n)

def lion_strike_damage(guards_w, traj_w):
    # Lion Fealty-Strike: A8 WS2+ S12 -4 D4 Lethal Hits vs T6 4++
    hits_raw = roll_d6(8)
    crit_hits = np.sum(hits_raw == 6)
    normal_hits = np.sum((hits_raw >= 2) & (hits_raw < 6))
    total_hits = normal_hits + crit_hits
    lethal_wounds = crit_hits  # auto-wound
    normal_wounds_roll = roll_d6(normal_hits)
    normal_wounds = np.sum(normal_wounds_roll >= 2)  # 2+
    total_wounds = lethal_wounds + normal_wounds
    save_fails = roll_d6(total_wounds)
    unsaved = np.sum(save_fails <= 3)  # 4++ fail 1-3
    dmg_rolls = roll_d6(unsaved)  # D4 as D6/1.75 approx but use D6 mean adjust? Wait D4: 1-4
    dmg = np.sum(np.clip(dmg_rolls, 1, 4))  # D4
    # Allocate to guards first
    dmg_dealt = 0
    for i in range(4):
        if guards_w[i] > 0:
            alloc = min(dmg, guards_w[i])
            guards_w[i] -= alloc
            dmg -= alloc
            dmg_dealt += alloc
            if dmg == 0:
                break
    if dmg > 0 and traj_w > 0:
        alloc = min(dmg, traj_w)
        traj_w -= alloc
        dmg_dealt += alloc
    return dmg_dealt, guards_w, traj_w

import numpy as np
np.random.seed(42)

def roll_dice(sides, n):
    return np.random.randint(1, sides + 1, n)

def d4_damage(n):
    return np.random.choice([1,2,3,4], n).sum()

def d3_damage(n):
    return np.random.choice([2,3,4], n).sum()

def lion_pistol_damage(guards_w, traj_w):
    hits = np.sum(roll_dice(6,4) >=2)  # BS2+
    wounds = np.sum(roll_dice(6,hits) >=4)  # S4 vs T6 4+
    unsaved = np.sum(roll_dice(6,wounds) <=3)  # 4++ fail 1-3
    dmg = 2 * unsaved
    dmg_left = dmg
    for i in range(4):
        if guards_w[i] > 0 and dmg_left > 0:
            alloc = min(dmg_left, guards_w[i])
            guards_w[i] -= alloc
            dmg_left -= alloc
    if dmg_left > 0 and traj_w > 0:
        alloc = min(dmg_left, traj_w)
        traj_w -= alloc
    return guards_w, traj_w

def lion_strike(guards_w, traj_w):
    hits_raw = roll_dice(6,8)
    crits = np.sum(hits_raw ==6)
    normals = np.sum((hits_raw >=2) & (hits_raw <6))
    lethal_wounds = crits
    normal_wounds = np.sum(roll_dice(6, normals) >=2)
    total_wounds = lethal_wounds + normal_wounds
    unsaved = np.sum(roll_dice(6, total_wounds) <=3)  # 4++
    dmg = d4_damage(unsaved)
    dmg_left = dmg
    for i in range(4):
        if guards_w[i] > 0 and dmg_left > 0:
            alloc = min(dmg_left, guards_w[i])
            guards_w[i] -= alloc
            dmg_left -= alloc
    if dmg_left > 0 and traj_w > 0:
        traj_w -= min(dmg_left, traj_w)
    return guards_w, traj_w, dmg

def cust_ranged_shoot(lion_w):
    # 4 Guards Spear ranged A2 BS2+ S4 -1 D2 vs T9 6+ no Vigil
    guard_hits = np.sum(roll_dice(6,8) >=2)
    guard_wounds_raw = roll_dice(6, guard_hits)
    guard_wounds = np.sum(guard_wounds_raw >=6)
    num1s = np.sum(guard_wounds_raw ==1)
    if num1s >0:
        rerolls = roll_dice(6, int(num1s))
        guard_wounds += np.sum(rerolls >=6)
    guard_unsaved = np.sum(roll_dice(6, guard_wounds) <=3)  # 3++
    guard_dmg = 2 * guard_unsaved
    # Traj Eagle A2 BS2+ S5 -2 D3 6+
    traj_hits = np.sum(roll_dice(6,2) >=2)
    traj_wounds = np.sum(roll_dice(6, traj_hits) >=6)
    traj_unsaved = np.sum(roll_dice(6, traj_wounds) <=3)
    traj_dmg = d3_damage(traj_unsaved)
    total_dmg = guard_dmg + traj_dmg
    lion_w -= total_dmg
    return lion_w

    guard_hits = np.sum(roll_dice(6, int(guard_att)) >=2)
    guard_wounds_raw = roll_dice(6, guard_hits)
    guard_wounds = np.sum(guard_wounds_raw >=5)  # 5+
    num1s = np.sum(guard_wounds_raw ==1)
    if num1s >0:
        rerolls = roll_dice(6, int(num1s))
        guard_wounds += np.sum(rerolls >=5)
    guard_unsaved = np.sum(roll_dice(6, guard_wounds) <=3)  # 3++
    guard_dmg = 2 * guard_unsaved
    # Traj Axe
    traj_att = 12 if moment_used[0] else 6
    traj_hits = np.sum(roll_dice(6, traj_att) >=2)
    traj_wounds = np.sum(roll_dice(6, traj_hits) >=4)  # 4+
    traj_unsaved = np.sum(roll_dice(6, traj_wounds) <=3)
    traj_dmg = d3_damage(traj_unsaved)
    total_dmg = guard_dmg + traj_dmg
    lion_w -= total_dmg
    moment_used[0] = True
    return lion_w

def run_single_battle():
    lion_w = 10.0
    traj_w = 7.0
    guards_w = np.array([3.0]*4)
    gap = 36.0
    rounds = 0
    engaged = False
    moment_used = [False]
    cust_total_initial = 19.0  # 12+7
    while lion_w > 0 and np.any(guards_w >0) or traj_w >0 and rounds <20:
        rounds +=1
        # Lion turn
        gap = max(0, gap - 8)
        if gap <=12:
            guards_w, traj_w = lion_pistol_damage(guards_w.copy(), traj_w)
        if not engaged and gap <=12:
            charge_roll = roll_dice(6,2).sum()
            if gap <= charge_roll:
                engaged = True
                # Fights First Strike
                guards_w, traj_w, _ = lion_strike(guards_w.copy(), traj_w)
        # Cust turn
        if not engaged:
            adv = 6 + roll_dice(6)
            gap = max(0, gap - adv)
            if gap <=24:
                lion_w = cust_ranged_shoot(lion_w)
            if gap <=12:
                charge_roll = roll_dice(6,2).sum()
                if gap <= charge_roll:
                    engaged = True
        if engaged:
            # Lion Fights First
            guards_w, traj_w, _ = lion_strike(guards_w.copy(), traj_w)
            if traj_w <=0 and np.all(guards_w <=0):
                break
            # Cust fight
            lion_w = cust_melee_fight(lion_w, guards_w, traj_w, moment_used)
            if lion_w <=0:
                break
    cust_remaining = np.sum(guards_w) + traj_w
    lion_dmg_dealt = cust_total_initial - cust_remaining
    cust_dmg_dealt = 10 - lion_w if lion_w <0 else 10 - lion_w
    if lion_w >0:
        return 'Lion', rounds, lion_dmg_dealt, cust_dmg_dealt
    else:
        return 'Cust', rounds, lion_dmg_dealt, cust_dmg_dealt

# Run 100
